name: Verify untrusted artifact as data
on:
  workflow_run:
    workflows: [Fork tests]
    types: [completed]

permissions:
  contents: read

jobs:
  inspect:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - uses: actions/download-artifact@3e5f45b2cfb9172054b4087a40e8e0b5a5461e7c # v8.0.1
        with:
          name: evidence
          path: untrusted-evidence
      - name: Verify declared digest; never execute artifact content
        working-directory: untrusted-evidence
        run: sha256sum --check SHA256SUMS
